package application_test
import (
 "context"
 "errors"
 "fmt"
 "testing"
 "time"
 . "github.com/kungrem23/quizgo/services/game/internal/application"
 game "github.com/kungrem23/quizgo/services/game/internal/domain"
)
func TestAuditDisconnectedPlayersExhaustLobby(t *testing.T) {
 snapshot:=game.QuizSnapshot{ID:1,Revision:1,Title:"audit",OwnerUserID:1,Questions:[]game.Question{{ID:1,Text:"Q",TimeLimitSeconds:10,Answers:[]game.Answer{{ID:1,Text:"A",IsCorrect:true},{ID:2,Text:"B"}}}}}
 repo:=&repositoryStub{};service:=New(catalogStub{snapshot:snapshot},repo,time.Hour)
 created,err:=service.CreateGame(context.Background(),CreateGameRequest{QuizID:1,AccessToken:"valid-test-token"});if err!=nil {t.Fatal(err)}
 hub:=NewHub(repo,time.Hour);defer hub.Close()
 for i:=0;i<100;i++ {
  joined,err:=hub.Join(context.Background(),created.Game.Code,fmt.Sprintf("bot-%d",i));if err!=nil {t.Fatal(err)}
  hub.Disconnect(joined.Session)
 }
 _,err=hub.Join(context.Background(),created.Game.Code,"real-player")
 if !errors.Is(err,game.ErrInvalidPlayer){t.Fatalf("unexpected join error: %v",err)}
 stored,err:=repo.GetByID(context.Background(),created.Game.ID);if err!=nil {t.Fatal(err)}
 if len(stored.Players)!=100 {t.Fatal("players not persisted")}
 t.Log("CONFIRMED: 100 sequential joins/disconnects fill the lobby; legitimate player rejected")
}
